--TABLES--

CREATE TABLE aeronaves (
id SERIAL PRIMARY KEY,
modelo VARCHAR(50) NOT NULL,
codigo_cauda VARCHAR(10) UNIQUE NOT NULL,
capacidade INT NOT NULL CHECK(capacidade >0)
);


CREATE TABLE pilotos (
id SERIAl PRIMARY KEY,
nome VARCHAR(150) NOT NULL,
codigo_anac VARCHAR(6) UNIQUE NOT NULL,
horas_voo INT DEFAULT 0 CHECK (horas_voo >= 0)
);


CREATE TABLE voos (
id SERIAL PRIMARY KEY,
aeronaves_id INT NOT NULL,
pilotos_id INT NOT NULL,
numero_voo VARCHAR(20) NOT NULL,
origem VARCHAR(50) NOT NULL,
destino VARCHAR(50) NOT NULL,
data_hora TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
status VARCHAR(150) DEFAULT ('AGENDADO'),

CONSTRAINT fk_voos_aeronaves
FOREIGN KEY (aeronaves_id)
REFERENCES aeronaves(id),


CONSTRAINT fk_voos_pilotos
FOREIGN KEY (pilotos_id)
REFERENCES pilotos(id),


CONSTRAINT check_status 
CHECK (status IN ('Agendado', 'Em Voo', 'Concluido', 'Cancelado'))
);


CREATE TABLE passageiros (
id SERIAL PRIMARY KEY, 
nome VARCHAR(100) NOT NULL, 
cpf VARCHAR(11) UNIQUE NOT NULL,
email VARCHAR(100) UNIQUE NOT NULL
)


CREATE TABLE passagens (
id SERIAL PRIMARY KEY,
voos_id INT NOT NULL,
passageiros_id INT NOT NULL,
assento VARCHAR(4) NOT NULL, 
classe VARCHAR(50) DEFAULT ('Economica'),
valor DECIMAL(10,2) NOT NULL CHECK (valor > 0),

CONSTRAINT fk_passagens_voos
FOREIGN KEY (voos_id)
REFERENCES voos(id),


CONSTRAINT fk_passagens_passageiros
FOREIGN KEY (passageiros_id)
REFERENCES passageiros(id),


CONSTRAINT check_classe
CHECK (classe IN ('Economica', 'Executiva'))
)
